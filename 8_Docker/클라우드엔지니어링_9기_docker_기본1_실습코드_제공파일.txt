실습6>  docker 이미지 목록 조회

docker images nginx		# repository
docker images nginx:latest    # repository:tag 

docker images -q                # 이미지 ID만 출력
				
docker images --no-trunc     # 정보를 줄이지 않고 전체 표시

docker images --digests        # 이미지 digest 표시 
					 
docker images --format json  # 출력 포맷 지정.  기본은 table
docker images --format table
										 	 
#https://docs.docker.com/reference/cli/docker/image/ls/#format  참조
docker images --format "table {{.ID}}\t{{.Repository}}\t{{.Tag}}"


docker images --filter "dangling=true"  # dangling 이미지만 조회.

docker images --filter "before=nginx"  # nginx 이미지보다 먼저 생성된 이미지 조회
docker images --filter "since=hello-world"  # hello-world 이미지보다 나중에 생성된 이미지 조회
docker images --filter=reference='nginx*' # nginx 전체 조회
docker images --filter=reference='*:latest' # latest 태그만 조회

실습7>  docker 이미지 상세 조회


docker image inspect nginx		                      # repository
docker image inspect nginx:latest	                      # repository:tag

docker images 
docker image inspect 1e4c05b897a7                     # 이미지 ID 지정 가능
				 
docker image inspect nginx  -f '{{.Id}}'                   # 이미지 ID 만 출력
docker image inspect   -f '{{.Id}}{{.Os}}'  nginx         # 이미지 ID와 Os 출력 
docker image inspect   -f '{{.Id}}******{{.Os}}'  nginx  # **** 를 구분자로 지정. 공백구분자도 가능

docker image inspect   -f '{{.Id}}******{{.Os}}'  nginx    # 계층적 구조
				
docker image inspect  nginx  hello-world  # 이미지 여러개 지정 가능.

실습10>  docker 이미지 삭제
			  
	
docker images 	
docker rmi  165ddde8751e    # TAG가 여러개인 경우 이미지 ID로 삭제시 에러남.
			
				
docker rmi nginx           # nginx:latest 와 동일, untagged	
docker rmi nginx:test      #  untagged
docker rmi nginx:v1        #  untagged

docker image rm nginx:v2  # 마지막 TAG이지만 컨테이너가 실행중이기 때문에 삭제 안됨					
docker ps -a			# nginx 컨테이너 모두 삭제 필요		
docker stop  d3e37a853f50    # 컨테이너 정상종료
docker rm    d3e37a853f50    # 컨테이너 삭제
docker rmi nginx:v2 		 # 컨테이너 삭제후 최종 nginx 이미지 삭제가능

실습13>  docker 이미지 업로드				  

1) hub.docker.com 에서 로그인하자. 
2) 저장소 생성.

오른쪽 상단 >  docker push 계정명  /my-app-repo:tagname  가이드됨.
			  			  
3) 이미지 태깅.
docker tag nginx:latest  계정명/my-app-repo:v1
					
4) 로그인  

docker login -u  이메일

# docker logout
				   
5) push
			  
docker push kyongyeolin/my-app-repo:v1
				  
6) 로컬이미지 삭제하고 pull 하자.

docker rmi    계정명/my-app-repo:v1
docker pull   계정명/my-app-repo:v1

실습19>  docker 컨테이너  목록 조회


docker ps			# 실행중인 컨테이너 목록
docker ps   -a 			# 모든 컨테이너 목록 (종료 포함)
docker ps  -q			# 현재 실행중인 컨테이너 ID만 출력
docker ps  -aq			# 모든 컨테이너 ID만 출력	
docker ps  -l			# 가장 최근에 생성된 컨테이너 1개 출력	
docker ps  -n 5			# 가장 최근에 생성된 컨테이너 5개 출력
docker ps --filter status=exited	# 종료된 컨테이너만 조회
docker ps  --filter status=running  # 실행 중 컨테이너 조회
docker ps --filter name=web	 #  name에 web이 포함된 컨테이너 조회
					
docker ps --format "{{.Names}}"	# 컨테이너의 이름만 출력
docker ps -q | wc -l 		# 현재 실행중인 컨테이너 개수 출력
