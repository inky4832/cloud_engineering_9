실습8>  boot_Dockerfile4_Network_mysql3_network.zip 이용 ( 컨테이너간 연동: 컨테이너명 이용 )
	
# 먼저 확인해야 될 사항.
Dockerfile :  multi-stage build 패턴 적용함.
build.gradle :   implementation("com.mysql:mysql-connector-j:8.3.0") # 버전 명시
application.yml :  url: jdbc:mysql://mysql-data3:3306/testdb     # MySQL 컨테이너명(ex.mysql-data3)명시
			
		
# network 생성
docker network create  boot-net
docker network ls

		   	
# mysql 생성	
docker volume create mysql-data-vol3
docker volume ls

docker run -it -d --name mysql-data3 -e MYSQL_ROOT_PASSWORD=1234  -e MYSQL_DATABASE=testdb  -v mysql-data-vol3:/var/lib/mysql  --network  boot-net  mysql:8.3


# boot 생성  
docker build -t spring-mysql-demo3 .
docker run -d --name spring-mysql-app3  -p 8090:8080  --network boot-net  spring-mysql-demo3
 
			  		  
#웹브라우저에서 요청:
http://localhost:8090/api/users
		

docker rm -f  spring-mysql-app3

실습9> project.zip 이용

#docker-compose.yml 내용

services:
  mysql-data5:
    image: mysql:8.3
    container_name: mysql-data5
    environment:
      MYSQL_ROOT_PASSWORD: 1234
      MYSQL_DATABASE: testdb
    volumes:
      - mysql-data-vol5:/var/lib/mysql
    networks:
      - fullstack-net

  springboot-backend-app:
    build:
      context: ./backend
      dockerfile: Dockerfile
    container_name: springboot-backend-app
    ports:
      - "8090:8080"
    depends_on:
      - mysql-data5
    networks:
      - fullstack-net

volumes:
  mysql-data-vol5:

networks:
  fullstack-net:
    driver: bridge

-------------------------------------------------------------------------------------------------	  
# compose 실행
			 
docker compose up -d  --build  # 백그라운드로 실행, 다시 빌드하기 
				        # --build 지정하지 않으면 이전 이미지 재사용함.
docker compose up   --build	


#웹브라우저에서 요청:
http://localhost:8090/api/users


docker compose down -v  # 컨테이너 종료 + volume 삭제 + network 삭제
