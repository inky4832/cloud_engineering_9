#!/bin/bash

# 배열생성
n=(10 20 30)

# 배열 조회1 
echo ${n[0]}     #  1			  			  
echo ${n[*]}     #  배열 전체 조회
echo ${n[@]}     #  배열 전체 조회
echo ${#n[@]}     #  배열 갯수
echo ${!n[@]}     #  배열 인덱스 조회

# 배열 추가
n[4]=40
n+=(50 60 70)
echo ${n[@]}       #  배열 전체 조회

# 배열 조회2
echo ${n[@]:1}     # 인덱스 1부터  배열 전체 조회
echo ${n[@]:1:3}   # 인덱스 1부터 3개 배열  조회


# 배열 삭제
unset n[0]
# unset n        #  배열 전체 삭제
echo ${n[@]}     #  배열 전체 조회


$ chmod 700 21.sh

#!/bin/bash

read -p "디렉터리 이름: " dirname

if [[ -z "$dirname" ]]; then
    echo "디렉터리 이름을 입력하세요."
    exit 1
fi

if [[ -d "$dirname" ]]; then
    echo "이미 존재하는 디렉터리입니다."
else
    mkdir "$dirname"
    echo "$dirname 디렉터리 생성 완료"
fi

#!/bin/bash

LOG="/var/log/auth.log"

if [[ ! -f "$LOG" ]]; then
    echo "로그 파일이 없습니다."
    exit 1
fi

echo "===== 로그인 실패 로그 분석 ====="

count=$(grep -c "authentication failure" "$LOG")

echo "인증 실패 횟수: $count"

echo "최근 인증 실패 내역:"

grep "authentication failure" "$LOG" | tail -5

#!/bin/bash

SOURCE="./dir1"
BACKUP="./tmp/backup"

TODAY=$(date +%Y%m%d)

mkdir -p "$BACKUP"

tar -czf "$BACKUP/dir1_$TODAY.tar.gz" "$SOURCE"

if [[ $? -eq 0 ]]; then
    echo "백업 성공"
else
    echo "백업 실패"
    exit 1
fi

echo "백업 파일 목록:"
ls -lh "$BACKUP"
