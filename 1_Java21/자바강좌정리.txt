논리연산자

논리값  && 논리값 : 그리고 해석 ( and 연산자 )
논리값  || 논리값 : 또는 해석  ( or 연산자 )
 !논리값         : 반대 해석  ( not 연산자 )
 

 and 논리식
 
 true && true:  true
 true && false: false
 false && true: false
 false && false: false
 
 
 
 or 논리식
 
 true || true:  true
 true || false: true
 false || true: true
 false || false: false
 
 not 논리식 
 !true: false
 !false: true

증감연산자(증가연산자: ++, 감소연산자: -- )

1)전치
  int n = 10;
  ++n;
  s.o.p(n) // 11
  
  //다른연산자와 같이 사용할 때
  int n2 = ++n; //전치 ( 먼저 증가하고 나중에 할당 )
  s.o.p(n2+" " + n) // 12  12

2)후치
  int n = 10;
  n++;
  s.o.p(n) //11

  //다른연산자와 같이 사용할 때
  int n2 = n++; //후치 ( 먼저  할당하고 나중에 증가 )
  s.o.p(n2+" " + n) // 11  12

반복문

예> hello 를 5번 출력?

   hello
   hello
   hello
   hello
   hello 
   
 3가지 구성요소가 필요함.
 1) 시작값
   ex. 1, 10, 100
   변수이용해서 시작값 작성함.
    int n = 1;   
  
 2) 증감연산자 
    ++n, --n
  
 3) 조건식 
    n < 6
   
 for문 ( 용도: 반복횟수가 예측가능한 경우)
  문법: 
    
   for(시작값; 조건식; 증감연산자){
   
   }
   ex. for(int n = 1; n < 6; ++n){
			System.out.println("hello");
        }
   
  while문 ( 용도: 반복횟수가 예측이 힘들때 또는 무한루프)
   문법:
   
      시작값;
      ...	  
      while(조건식){
	    ...
	    증감연산자
		...
	  }
   ex.
        int n = 1;
		 while(n < 6){
	    System.out.println("hello");
	    ++n;
	  }
	  
  do~while문 ( 용도: 조건이 일치하지 않아도 반드시 1번은 실행됨)
  
  문법:
  
     시작값;
	 ..
     do{
	    ...
	    증감연산자
		...
	 }while(조건식);
	 
  ex.
       int n = 1;
	   do{
	    System.out.println("hello");
	    ++n;
	   }while(n < 6);
	   
 break문 ( 용도: 반복문을 빠져나올때 )
 
 문법:
     
	   for(시작값; 조건식; 증감연산자){
          ..
          if(조건식)break;
		  ...
	   }
	   System.out.end("End");
	   
 continue문 ( 용도: 반복해야될 특정 문장을 skip할 목적)
 
 문법:
     
	   for(시작값; 조건식; 증감연산자){
          문장1;
          if(조건식)continue;
		  문장2;
		  문장3;
	   }
	   System.out.end("End");
	   
변수 scope 
     
	   for(시작값; 조건식; 증감연산자){
           int n=10; // for블럭내에서만 사용가능
	   }
	   System.out.end(n);  //컴파일에러
	   
	   if(조건식){
	       int n2=10; // if블럭내에서만 사용가능
	   }
	    System.out.end(n2);  //컴파일에러
		
	==> 해결
	    int n2=0
	   if(조건식){
	       n2=10; // if블럭내에서만 사용가능
	   }
	    System.out.end(n2);  //출력
